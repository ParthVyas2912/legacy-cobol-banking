1. MOVE COMM-ACCT-NUMBER TO ACCT-NUMBER
2. MOVE ZEROES TO ACCT-BALANCE
3. MOVE 'A' TO ACCT-STATUS
4. EXEC CICS  WRITE FILE('ACCTFILE')
                   FROM(ACCT-RECORD)
                   RIDFLD(ACCT-NUMBER)
                   RESP(WS-RESPONSE-CODE)
     END-EXEC
5. IF WS-RESPONSE-CODE = 0 EXEC SQL  INSERT INTO ACCT_AUDIT
                   (ACCT_NUM, ACTION, ACTOR, TS)
                VALUES (:ACCT-NUMBER, 'CREATE',
                        :COMM-FUNCTION, CURRENT_TIMESTAMP)
         END-EXEC MOVE 0 TO COMM-RESULT-CODE ELSE MOVE 1 TO COMM-RESULT-CODE END-IF
