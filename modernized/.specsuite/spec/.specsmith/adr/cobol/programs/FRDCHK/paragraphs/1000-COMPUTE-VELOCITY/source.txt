1. EXEC SQL         SELECT COUNT(*), COALESCE(SUM(AMOUNT),0)
          INTO :WS-VELOCITY-COUNT, :WS-VELOCITY-AMT

         WHERE ACCT = :TXN-ACCT-NUMBER
           AND POSTED_AT >= CURRENT_TIMESTAMP - 1 HOUR
     END-EXEC
